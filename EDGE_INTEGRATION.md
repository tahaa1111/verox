## 2. Authentication

The Pi must obtain a Firebase ID token for every API call. There are two paths to acquire one.

### Path A — Workload Identity Federation (v2 hardening — optional)

No long-lived key file on disk. The Pi proves its identity via a hardware-backed credential (device serial, TPM attestation, or X.509 client certificate), exchanges it with GCP Secure Token Service (STS) for a short-lived Federated Access Token, then calls a lightweight Cloud Function to mint a Firebase custom token, and finally signs in to Firebase to get the ID token.

**This is not the default path.** Enable it only after the deployment is stable and working with keyfile auth. To set it up, run `bash scripts/05a_setup_wif_optional.sh` on the deploy machine.

This path requires the WIF credential configuration file generated by `05a_setup_wif_optional.sh` and stored at `MEDIBOX_CREDENTIALS_PATH`.

```python
# auth_wif.py
import os, time, requests
import google.auth
import google.auth.transport.requests
from google.oauth2 import credentials as oauth2_credentials

CREDENTIALS_PATH = os.environ["MEDIBOX_CREDENTIALS_PATH"]   # credential config JSON
FIREBASE_API_KEY = os.environ["MEDIBOX_FIREBASE_API_KEY"]   # public key from Firebase Console
CUSTOM_TOKEN_URL = os.environ["MEDIBOX_API_URL"] + "/v1/auth/custom-token"

_cached: dict = {}   # {"id_token": str, "expires_at": float}


def _gcp_access_token() -> str:
    """Exchange the WIF credential config for a short-lived GCP access token."""
    creds, _ = google.auth.load_credentials_from_file(
        CREDENTIALS_PATH,
        scopes=["https://www.googleapis.com/auth/cloud-platform"],
    )
    req = google.auth.transport.requests.Request()
    creds.refresh(req)
    return creds.token


def _firebase_custom_token(gcp_access_token: str) -> str:
    """Call the medibox API to mint a Firebase custom token for this device."""
    device_id = os.environ["MEDIBOX_DEVICE_ID"]
    resp = requests.post(
        CUSTOM_TOKEN_URL,
        json={"device_id": device_id},
        headers={"Authorization": f"Bearer {gcp_access_token}"},
        timeout=10,
    )
    resp.raise_for_status()
    return resp.json()["custom_token"]


def _sign_in_with_custom_token(custom_token: str) -> tuple[str, float]:
    """Exchange a Firebase custom token for an ID token + expiry."""
    url = (
        "https://identitytoolkit.googleapis.com/v1/accounts:signInWithCustomToken"
        f"?key={FIREBASE_API_KEY}"
    )
    resp = requests.post(
        url,
        json={"token": custom_token, "returnSecureToken": True},
        timeout=10,
    )
    resp.raise_for_status()
    data = resp.json()
    expires_at = time.time() + int(data["expiresIn"]) - 300  # 5 min safety margin
    return data["idToken"], expires_at


def get_id_token() -> str:
    """Return a valid Firebase ID token, refreshing if within 5 min of expiry."""
    if _cached.get("id_token") and time.time() < _cached.get("expires_at", 0):
        return _cached["id_token"]
    gcp_tok = _gcp_access_token()
    custom_tok = _firebase_custom_token(gcp_tok)
    id_token, expires_at = _sign_in_with_custom_token(custom_tok)
    _cached["id_token"] = id_token
    _cached["expires_at"] = expires_at
    return id_token
```

```python
# auth_keyfile.py
import os, time, json, requests
import google.auth
import google.auth.transport.requests
from google.oauth2 import service_account

CREDENTIALS_PATH = os.environ["MEDIBOX_CREDENTIALS_PATH"]
FIREBASE_API_KEY = os.environ["MEDIBOX_FIREBASE_API_KEY"]

_cached: dict = {}


def _sa_access_token() -> str:
    creds = service_account.Credentials.from_service_account_file(
        CREDENTIALS_PATH,
        scopes=["https://www.googleapis.com/auth/cloud-platform"],
    )
    req = google.auth.transport.requests.Request()
    creds.refresh(req)
    return creds.token


def _firebase_custom_token_via_admin_rest(sa_access_token: str) -> str:
    """
    Use the Firebase Auth REST API to create a custom token.
    Requires the SA to have roles/firebase.sdkAdminServiceAgent or
    roles/iam.serviceAccountTokenCreator on the Firebase Admin SA.
    """
    device_id = os.environ["MEDIBOX_DEVICE_ID"]
    project_id = json.load(open(CREDENTIALS_PATH))["project_id"]
    url = (
        f"https://identitytoolkit.googleapis.com/v1/projects/{project_id}"
        f"/tenants/-/accounts:signInWithCustomToken"
    )
    # simpler: call the medibox API's /v1/auth/custom-token endpoint
    resp = requests.post(
        os.environ["MEDIBOX_API_URL"] + "/v1/auth/custom-token",
        json={"device_id": device_id},
        headers={"Authorization": f"Bearer {sa_access_token}"},
        timeout=10,
    )
    resp.raise_for_status()
    return resp.json()["custom_token"]


def _sign_in_with_custom_token(custom_token: str) -> tuple[str, float]:
    url = (
        "https://identitytoolkit.googleapis.com/v1/accounts:signInWithCustomToken"
        f"?key={FIREBASE_API_KEY}"
    )
    resp = requests.post(
        url, json={"token": custom_token, "returnSecureToken": True}, timeout=10
    )
    resp.raise_for_status()
    data = resp.json()
    expires_at = time.time() + int(data["expiresIn"]) - 300
    return data["idToken"], expires_at


def get_id_token() -> str:
    if _cached.get("id_token") and time.time() < _cached.get("expires_at", 0):
        return _cached["id_token"]
    sa_tok = _sa_access_token()
    custom_tok = _firebase_custom_token_via_admin_rest(sa_tok)
    id_token, expires_at = _sign_in_with_custom_token(custom_tok)
    _cached["id_token"] = id_token
    _cached["expires_at"] = expires_at
    return id_token
```

### POST /v1/submit

Submits one prescription session (1–30 JPEG/PNG crop images).

**Request** — `multipart/form-data`

| Field | Type | Required | Description |
|-------|------|----------|-------------|
| `device_id` | string | yes | e.g. `pi-0001` |
| `session_id` | string | yes | UUID generated on the Pi per prescription event |
| `crops` | file[] | yes | 1–30 JPEG or PNG files, each ≤ 2 MB |

```python
# submit_example.py
import os, uuid, httpx
from auth import get_id_token
from pathlib import Path

API_URL = os.environ["MEDIBOX_API_URL"]
DEVICE_ID = os.environ["MEDIBOX_DEVICE_ID"]


def submit_crops(crop_paths: list[Path]) -> str:
    """Returns job_id on success. Raises httpx.HTTPStatusError on failure."""
    token = get_id_token()
    files = [
        ("crops", (p.name, p.read_bytes(), "image/jpeg")) for p in crop_paths
    ]
    data = {"device_id": DEVICE_ID, "session_id": str(uuid.uuid4())}
    resp = httpx.post(
        f"{API_URL}/v1/submit",
        headers={"Authorization": f"Bearer {token}"},
        files=files,
        data=data,
        timeout=30,
    )
    resp.raise_for_status()
    return resp.json()["job_id"]
```

**Response 202** (accepted):
```json
{
  "job_id": "3fa85f64-5717-4562-b3fc-2c963f66afa6",
  "status": "pending",
  "message": "Pharmacist verification required. Medibox assists, it does not dispense."
}
```

**Error responses:**

| Code | Meaning | Action |
|------|---------|--------|
| 400 | Missing field or bad content-type | Fix payload, do not retry |
| 401 | Token expired or invalid | Refresh token (`get_id_token(force=True)`), retry once |
| 403 | Device not authorised | Alert operator |
| 413 | Total payload > 20 MB | Split into ≤ 15 crops per call |
| 422 | Validation failure (e.g. 0 crops sent) | Log details, do not retry |
| 429 | Rate limited | Exponential backoff — see §4 |
| 500 | Server error | Retry with backoff, max 3 times |
| 503 | Cloud Run cold-starting | Retry with backoff, max 10 min |

### GET /v1/result/{job_id}

Poll for a completed result. Preferred path is WebSocket (§ below); use this as fallback.

```python
def poll_result(job_id: str, timeout_s: int = 120) -> dict:
    import time
    token = get_id_token()
    deadline = time.time() + timeout_s
    interval = float(os.environ.get("MEDIBOX_POLL_INTERVAL_SECONDS", "2"))
    max_wait = float(os.environ.get("MEDIBOX_POLL_MAX_WAIT_SECONDS", "30"))

    while time.time() < deadline:
        resp = httpx.get(
            f"{API_URL}/v1/result/{job_id}",
            headers={"Authorization": f"Bearer {token}"},
            timeout=10,
        )
        if resp.status_code == 404:
            time.sleep(interval)
            interval = min(interval * 1.5, max_wait)
            continue
        resp.raise_for_status()
        data = resp.json()
        if data["status"] in ("completed", "failed"):
            return data
        time.sleep(interval)
        interval = min(interval * 1.5, max_wait)

    raise TimeoutError(f"Job {job_id} did not complete within {timeout_s}s")
```

**Response 200** (completed):

### WSS /v1/ws/jobs/{job_id}

Receive real-time status updates. The server sends JSON events; the connection closes when the job reaches a terminal state (`completed` or `failed`).

**Event types:**

| `type` | When sent | Payload fields |
|--------|-----------|---------------|
| `status_update` | Each state transition | `status`, `message` |
| `result` | Job completed | `status`, `result`, `disclaimer` |
| `error` | Job failed | `status`, `error`, `disclaimer` |
| `ping` | Every 15 s (keepalive) | — |

```python
# ws_listener.py
import asyncio, json, os, time
import websockets
from auth import get_id_token

API_URL = os.environ["MEDIBOX_API_URL"]
# Replace https:// with wss://
WS_BASE = API_URL.replace("https://", "wss://").replace("http://", "ws://")


async def watch_job(job_id: str, on_result) -> dict:
    """
    Connects via WebSocket and returns the result dict when the job is terminal.
    Falls back to HTTP polling if WebSocket fails to connect or drops repeatedly.
    """
    token = get_id_token()
    uri = f"{WS_BASE}/v1/ws/jobs/{job_id}?token={token}"

    backoff = 1
    for attempt in range(5):
        try:
            async with websockets.connect(uri, ping_interval=20, ping_timeout=15) as ws:
                backoff = 1  # reset on successful connect
                async for raw in ws:
                    event = json.loads(raw)
                    if event.get("type") == "ping":
                        continue
                    on_result(event)            # callback to update UI / queue
                    if event.get("status") in ("completed", "failed"):
                        return event
        except (websockets.ConnectionClosed, OSError) as exc:
            if attempt == 4:
                break   # fall through to HTTP poll
            await asyncio.sleep(backoff)
            backoff = min(backoff * 2, 30)
            token = get_id_token()              # token may have expired
            uri = f"{WS_BASE}/v1/ws/jobs/{job_id}?token={token}"

    # WebSocket persistently failed — fall back to HTTP polling
    import httpx
    return poll_result(job_id)  # from result polling example above
```

## 4. Error Handling and Retry Policy

| HTTP code | Meaning | Recommended action |
|-----------|---------|-------------------|
| 200 | Success | Continue |
| 202 | Accepted, processing | Poll `/result/{id}` or listen on WS |
| 400 | Bad payload | Log, do not retry — check schema |
| 401 | Token expired or invalid | Refresh token, retry once |
| 403 | Forbidden | Log, alert operator, do not retry |
| 413 | Payload too large | Split crops into multiple submissions |
| 422 | Validation failed | Log details, do not retry |
| 429 | Rate limited | Exponential backoff: 1 s → 2 s → 4 s → 8 s → max 60 s |
| 500 | Server error | Retry with backoff, max 3 times |
| 502 / 503 | Cloud Run cold-starting or unavailable | Retry with backoff, max 10 min before alerting |
| 504 | Gateway timeout | Retry with backoff once |
| Network error | Connection refused or timeout | Queue locally, retry every 30 s, alert after 1 h |

```python
# retry.py
import time, httpx

def with_retry(fn, *, max_attempts: int = 3, retryable_codes=(429, 500, 502, 503, 504)):
    """
    Wraps a callable that returns an httpx.Response.
    Raises on non-retryable errors or after max_attempts.
    """
    backoff = 1.0
    for attempt in range(max_attempts):
        try:
            resp = fn()
            if resp.status_code not in retryable_codes:
                resp.raise_for_status()
                return resp
            if attempt == max_attempts - 1:
                resp.raise_for_status()
            time.sleep(min(backoff, 60))
            backoff *= 2
        except (httpx.ConnectError, httpx.TimeoutException):
            if attempt == max_attempts - 1:
                raise
            time.sleep(min(backoff, 60))
            backoff *= 2
    raise RuntimeError("Unreachable")
```

## 5. Local Offline Queue

When the API is unreachable the Pi must not drop prescriptions. The queue persists jobs to SQLite so they survive reboots.

```python
# submission_queue.py
"""
SQLite-backed offline queue for Medibox crop submissions.

Schema
------
jobs(
  id            INTEGER PRIMARY KEY AUTOINCREMENT,
  session_id    TEXT    NOT NULL UNIQUE,
  crop_dir      TEXT    NOT NULL,   -- absolute path to directory of JPEG files
  device_id     TEXT    NOT NULL,
  status        TEXT    NOT NULL DEFAULT 'pending',
                        -- pending | submitted | confirmed | failed
  job_id        TEXT,               -- filled in after successful submit
  error         TEXT,
  created_at    REAL    NOT NULL,   -- Unix timestamp
  updated_at    REAL    NOT NULL
)
"""
import os, time, json, logging, sqlite3, uuid
from pathlib import Path
from contextlib import contextmanager

DB_PATH = os.environ.get("MEDIBOX_OFFLINE_QUEUE_PATH", "/var/lib/medibox/queue.db")
MAX_QUEUE = int(os.environ.get("MEDIBOX_MAX_OFFLINE_QUEUE", "500"))
CONFIRM_TIMEOUT_S = 300   # 5 min — if submitted but no result, re-queue
RETENTION_DAYS = 7

log = logging.getLogger(__name__)


@contextmanager
def _conn():
    con = sqlite3.connect(DB_PATH, check_same_thread=False)
    con.row_factory = sqlite3.Row
    try:
        yield con
        con.commit()
    finally:
        con.close()


def init_db():
    Path(DB_PATH).parent.mkdir(parents=True, exist_ok=True)
    with _conn() as con:
        con.execute("""
            CREATE TABLE IF NOT EXISTS jobs (
                id          INTEGER PRIMARY KEY AUTOINCREMENT,
                session_id  TEXT    NOT NULL UNIQUE,
                crop_dir    TEXT    NOT NULL,
                device_id   TEXT    NOT NULL,
                status      TEXT    NOT NULL DEFAULT 'pending',
                job_id      TEXT,
                error       TEXT,
                created_at  REAL    NOT NULL,
                updated_at  REAL    NOT NULL
            )
        """)
        con.execute("CREATE INDEX IF NOT EXISTS idx_status ON jobs(status)")


def enqueue(crop_dir: str, device_id: str) -> str:
    """Add a new pending submission. Returns session_id."""
    pending_count = _count_by_status("pending")
    if pending_count >= MAX_QUEUE:
        raise OverflowError(
            f"Offline queue full ({pending_count} pending). "
            "Check network connectivity."
        )
    session_id = str(uuid.uuid4())
    now = time.time()
    with _conn() as con:
        con.execute(
            "INSERT INTO jobs (session_id, crop_dir, device_id, created_at, updated_at) "
            "VALUES (?, ?, ?, ?, ?)",
            (session_id, crop_dir, device_id, now, now),
        )
    log.info("Queued session %s (%s)", session_id, crop_dir)
    return session_id


def _count_by_status(status: str) -> int:
    with _conn() as con:
        row = con.execute(
            "SELECT COUNT(*) FROM jobs WHERE status=?", (status,)
        ).fetchone()
        return row[0]


def _get_pending() -> list[sqlite3.Row]:
    with _conn() as con:
        return con.execute(
            "SELECT * FROM jobs WHERE status='pending' ORDER BY created_at LIMIT 10"
        ).fetchall()


def _get_stale_submitted() -> list[sqlite3.Row]:
    cutoff = time.time() - CONFIRM_TIMEOUT_S
    with _conn() as con:
        return con.execute(
            "SELECT * FROM jobs WHERE status='submitted' AND updated_at < ?",
            (cutoff,),
        ).fetchall()


def _update(session_id: str, **kwargs):
    kwargs["updated_at"] = time.time()
    cols = ", ".join(f"{k}=?" for k in kwargs)
    with _conn() as con:
        con.execute(
            f"UPDATE jobs SET {cols} WHERE session_id=?",
            (*kwargs.values(), session_id),
        )


def purge_old():
    cutoff = time.time() - RETENTION_DAYS * 86400
    with _conn() as con:
        n = con.execute(
            "DELETE FROM jobs WHERE status='confirmed' AND created_at < ?",
            (cutoff,),
        ).rowcount
    if n:
        log.info("Purged %d confirmed jobs older than %d days", n, RETENTION_DAYS)


# ---------------------------------------------------------------------------
# Submitter loop — call this in a background thread
# ---------------------------------------------------------------------------

def submitter_loop(submit_fn, poll_fn, stop_event):
    """
    submit_fn(crop_dir, session_id, device_id) -> job_id   (raises on failure)
    poll_fn(job_id) -> dict with keys: status, result/error
    stop_event: threading.Event — set it to exit the loop
    """
    import threading
    log.info("Submitter loop started")
    while not stop_event.is_set():
        # Re-queue stale submitted jobs
        for row in _get_stale_submitted():
            log.warning("Re-queuing stale job %s (submitted but not confirmed)", row["session_id"])
            _update(row["session_id"], status="pending")

        # Attempt to submit pending jobs
        for row in _get_pending():
            crop_dir = row["crop_dir"]
            session_id = row["session_id"]
            try:
                job_id = submit_fn(crop_dir, session_id, row["device_id"])
                _update(session_id, status="submitted", job_id=job_id)
                log.info("Submitted %s → job_id=%s", session_id, job_id)

                # Immediately try to confirm (non-blocking poll)
                threading.Thread(
                    target=_confirm_job,
                    args=(session_id, job_id, poll_fn),
                    daemon=True,
                ).start()

            except OverflowError:
                log.error("Queue full — cannot submit %s", session_id)
            except Exception as exc:
                log.warning("Submit failed for %s: %s", session_id, exc)

        # Health check alert
        pending = _count_by_status("pending")
        if pending > 100:
            log.error(
                "ALERT: %d jobs pending. Network may be down for > 1h.", pending
            )

        purge_old()
        stop_event.wait(30)   # retry interval


def _confirm_job(session_id: str, job_id: str, poll_fn):
    try:
        result = poll_fn(job_id)
        if result["status"] == "completed":
            _update(session_id, status="confirmed")
            log.info("Confirmed %s (job_id=%s)", session_id, job_id)
        else:
            _update(session_id, status="failed", error=result.get("error", "unknown"))
            log.warning("Job %s failed: %s", job_id, result.get("error"))
    except Exception as exc:
        log.warning("Could not confirm %s: %s — will retry", job_id, exc)
        # Status remains 'submitted'; stale-check loop will re-queue after CONFIRM_TIMEOUT_S
```

**Starting the loop in your main application:**

```python

    args=(my_submit_fn, my_poll_fn, stop),
    daemon=True,
)
t.start()
```


### Periodic health probe with LED indicator

```python
# health_probe.py — run in a background thread every 5 min
import time, logging
try:
    from gpiozero import LED
    _led = LED(17)   # GPIO 17, adjust to your wiring
except Exception:
    _led = None      # not on Pi or gpiozero not installed

log = logging.getLogger(__name__)

def health_loop(check_health_fn, stop_event, interval_s: int = 300):
    while not stop_event.wait(interval_s):
        ok = check_health_fn()
        if _led:
            if ok:
                _led.on()
            else:
                _led.blink(on_time=0.2, off_time=0.2, n=5)
        if not ok:
            log.error("Health probe failed — API unreachable")
```

## 9. Migration Path: Keyfile → WIF

When WIF becomes available (e.g., the GCP org policy is updated or a TPM-based attestation is configured), you can migrate without touching any application code.

**Steps:**

1. Generate the WIF credential configuration file using `01_setup_project.sh` or manually:
   ```bash
   gcloud iam workload-identity-pools create-cred-config \
     "projects/<PROJECT_NUMBER>/locations/global/workloadIdentityPools/medibox-edge-pool/providers/medibox-edge-provider" \
     --service-account="medibox-edge@<PROJECT_ID>.iam.gserviceaccount.com" \
     --output-file=/etc/medibox/medibox-wif-config.json \
     --credential-source-file=/run/medibox/device-token \
     --credential-source-type=text
   ```

2. Copy the new config to the Pi:
   ```bash
   sudo cp medibox-wif-config.json /etc/medibox/medibox-edge.json
   sudo chown medibox:medibox /etc/medibox/medibox-edge.json
   sudo chmod 600 /etc/medibox/medibox-edge.json
   ```

3. Update two lines in `/etc/medibox/edge.env`:
   ```bash
   MEDIBOX_AUTH_MODE=wif
   MEDIBOX_CREDENTIALS_PATH=/etc/medibox/medibox-edge.json
   ```

4. Restart the service:
   ```bash
   sudo systemctl restart medibox-edge
   journalctl -u medibox-edge -n 20
   ```

The `auth.py` module's runtime switch (`if AUTH_MODE == "wif"`) handles the rest. No code change is needed on the Pi.

**Why bother?** WIF eliminates a long-lived credential on disk. If the Pi is stolen or compromised, there is no exportable key — only a device-bound attestation that cannot be used from another machine.
