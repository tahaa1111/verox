"""
Auth via service account JSON keyfile.
Gets a GCP access token, exchanges it for a Firebase custom token via the
medibox API, then signs into Firebase to get an ID token for API calls.
"""
import os, time, json, logging, requests
from google.oauth2 import service_account
import google.auth.transport.requests

CREDENTIALS_PATH = os.environ["MEDIBOX_CREDENTIALS_PATH"]
FIREBASE_API_KEY  = os.environ["MEDIBOX_FIREBASE_API_KEY"]
API_URL           = os.environ["MEDIBOX_API_URL"]
DEVICE_ID         = os.environ["MEDIBOX_DEVICE_ID"]

log = logging.getLogger(__name__)
_cache: dict = {}


def _gcp_access_token() -> str:
    creds = service_account.Credentials.from_service_account_file(
        CREDENTIALS_PATH,
        scopes=["https://www.googleapis.com/auth/cloud-platform"],
    )
    creds.refresh(google.auth.transport.requests.Request())
    return creds.token


def _firebase_custom_token(gcp_token: str) -> str:
    resp = requests.post(
        f"{API_URL}/v1/auth/custom-token",
        json={"device_id": DEVICE_ID},
        headers={"Authorization": f"Bearer {gcp_token}"},
        timeout=15,
    )
    resp.raise_for_status()
    return resp.json()["custom_token"]


def _sign_in(custom_token: str) -> tuple[str, float]:
    url = ("https://identitytoolkit.googleapis.com/v1/accounts:signInWithCustomToken"
           f"?key={FIREBASE_API_KEY}")
    resp = requests.post(
        url, json={"token": custom_token, "returnSecureToken": True}, timeout=15)
    resp.raise_for_status()
    data = resp.json()
    expires_at = time.time() + int(data["expiresIn"]) - 300  # refresh 5 min early
    return data["idToken"], expires_at


def get_id_token(force: bool = False) -> str:
    """Return a valid Firebase ID token, refreshing automatically near expiry."""
    if not force and _cache.get("id_token") and time.time() < _cache.get("expires_at", 0):
        return _cache["id_token"]
    log.info("Refreshing Firebase ID token")
    gcp_tok    = _gcp_access_token()
    custom_tok = _firebase_custom_token(gcp_tok)
    id_token, expires_at = _sign_in(custom_tok)
    _cache["id_token"]   = id_token
    _cache["expires_at"] = expires_at
    return id_token
PYEOF

"""
SQLite offline queue for crop submissions.
Survives reboots and network outages.
"""
import os, time, uuid, logging, sqlite3, threading
from pathlib import Path
from contextlib import contextmanager

DB_PATH   = os.environ.get("MEDIBOX_OFFLINE_QUEUE_PATH", "/var/lib/medibox/queue.db")
MAX_QUEUE = int(os.environ.get("MEDIBOX_MAX_OFFLINE_QUEUE", "500"))
CONFIRM_TIMEOUT = 300   # seconds before a submitted-but-unconfirmed job is re-queued
RETENTION_DAYS  = 7

log = logging.getLogger(__name__)


@contextmanager
def _conn():
    con = sqlite3.connect(DB_PATH, check_same_thread=False)
    con.row_factory = sqlite3.Row
    try:
        yield con
        con.commit()
    finally:
        con.close()


def init_db():
    Path(DB_PATH).parent.mkdir(parents=True, exist_ok=True)
    with _conn() as c:
        c.execute("""
            CREATE TABLE IF NOT EXISTS jobs (
                id         INTEGER PRIMARY KEY AUTOINCREMENT,
                session_id TEXT    NOT NULL UNIQUE,
                crop_dir   TEXT    NOT NULL,
                device_id  TEXT    NOT NULL,
                status     TEXT    NOT NULL DEFAULT 'pending',
                job_id     TEXT,
                error      TEXT,
                created_at REAL    NOT NULL,
                updated_at REAL    NOT NULL
            )
        """)
        c.execute("CREATE INDEX IF NOT EXISTS idx_status ON jobs(status)")


def enqueue(crop_dir: str, device_id: str) -> str:
    pending = _count("pending")
    if pending >= MAX_QUEUE:
        raise OverflowError(f"Queue full ({pending} pending)")
    sid = str(uuid.uuid4())
    now = time.time()
    with _conn() as c:
        c.execute(
            "INSERT INTO jobs (session_id,crop_dir,device_id,created_at,updated_at) "
            "VALUES (?,?,?,?,?)", (sid, crop_dir, device_id, now, now))
    log.info("Queued %s (%s)", sid, crop_dir)
    return sid


def _count(status):
    with _conn() as c:
        return c.execute("SELECT COUNT(*) FROM jobs WHERE status=?", (status,)).fetchone()[0]


def _update(session_id, **kw):
    kw["updated_at"] = time.time()
    cols = ", ".join(f"{k}=?" for k in kw)
    with _conn() as c:
        c.execute(f"UPDATE jobs SET {cols} WHERE session_id=?", (*kw.values(), session_id))


def _pending():
    with _conn() as c:
        return c.execute(
            "SELECT * FROM jobs WHERE status='pending' ORDER BY created_at LIMIT 10"
        ).fetchall()


def _stale_submitted():
    cutoff = time.time() - CONFIRM_TIMEOUT
    with _conn() as c:
        return c.execute(
            "SELECT * FROM jobs WHERE status='submitted' AND updated_at<?", (cutoff,)
        ).fetchall()


def purge_old():
    cutoff = time.time() - RETENTION_DAYS * 86400
    with _conn() as c:
        n = c.execute("DELETE FROM jobs WHERE status='confirmed' AND created_at<?",
                      (cutoff,)).rowcount
    if n:
        log.info("Purged %d old confirmed jobs", n)


def _confirm(session_id, job_id, poll_fn):
    try:
        result = poll_fn(job_id)
        if result.get("status") == "completed":
            _update(session_id, status="confirmed")
            log.info("Confirmed %s", job_id)
        else:
            _update(session_id, status="failed", error=result.get("error", "unknown"))
    except Exception as exc:
        log.warning("Could not confirm %s: %s", job_id, exc)


def submitter_loop(submit_fn, poll_fn, stop_event):
    """Run in a background thread. submit_fn and poll_fn are callables."""
    log.info("Submitter loop started")
    while not stop_event.is_set():
        for row in _stale_submitted():
            log.warning("Re-queuing stale job %s", row["session_id"])
            _update(row["session_id"], status="pending")

        for row in _pending():
            try:
                job_id = submit_fn(row["crop_dir"], row["session_id"], row["device_id"])
                _update(row["session_id"], status="submitted", job_id=job_id)
                threading.Thread(
                    target=_confirm, args=(row["session_id"], job_id, poll_fn),
                    daemon=True).start()
            except Exception as exc:
                log.warning("Submit failed for %s: %s", row["session_id"], exc)

        if _count("pending") > 100:
            log.error("ALERT: >100 jobs pending — check network/API connectivity")

        purge_old()
        stop_event.wait(30)
PYEOF

"""
Medibox edge client — Raspberry Pi 5
Captures a prescription image, slices into a 3x3 grid of crops,
submits to the cloud API, and prints the structured result.

Usage:
  /opt/medibox/venv/bin/python /opt/medibox/submit_prescription.py
  /opt/medibox/venv/bin/python /opt/medibox/submit_prescription.py --image /path/to/file.jpg
"""
import os, sys, uuid, time, json, logging, argparse, tempfile
from pathlib import Path
import httpx
from PIL import Image
from auth import get_id_token
from submission_queue import init_db, enqueue, submitter_loop
import threading

# ── Config from env ────────────────────────────────────────────────────────
API_URL   = os.environ["MEDIBOX_API_URL"]
DEVICE_ID = os.environ["MEDIBOX_DEVICE_ID"]
POLL_INTERVAL = int(os.environ.get("MEDIBOX_POLL_INTERVAL_SECONDS", "2"))
POLL_MAX_WAIT = int(os.environ.get("MEDIBOX_POLL_MAX_WAIT_SECONDS", "120"))

GRID_ROWS  = 3
GRID_COLS  = 3
CROP_SIZE  = 1536       # pixels per crop side
MAX_BYTES  = 2_000_000  # 2 MB per crop

log = logging.getLogger(__name__)


# ── Camera capture ─────────────────────────────────────────────────────────
def capture_image(out_path: str):
    try:
        from picamera2 import Picamera2
        cam = Picamera2()
        cam.configure(cam.create_still_configuration())
        cam.start()
        time.sleep(2)  # warm-up
        cam.capture_file(out_path)
        cam.stop()
        log.info("Captured image to %s", out_path)
    except Exception as exc:
        log.error("Camera capture failed: %s", exc)
        raise


# ── Grid slicer ────────────────────────────────────────────────────────────
def slice_into_crops(image_path: str, out_dir: str) -> list[str]:
    img = Image.open(image_path).convert("RGB")
    w, h = img.size
    if w * h > 25_000_000:
        raise ValueError(f"Image too large ({w}x{h}) — max 25 MP")

    cell_w = w // GRID_COLS
    cell_h = h // GRID_ROWS
    paths  = []

    for row in range(GRID_ROWS):
        for col in range(GRID_COLS):
            left   = col * cell_w
            upper  = row * cell_h
            right  = left + cell_w
            lower  = upper + cell_h
            crop   = img.crop((left, upper, right, lower)).resize(
                         (CROP_SIZE, CROP_SIZE), Image.LANCZOS)
            path   = os.path.join(out_dir, f"crop_{row}{col}.jpg")
            crop.save(path, "JPEG", quality=82)
            # Re-compress if still over 2 MB
            if os.path.getsize(path) > MAX_BYTES:
                crop.save(path, "JPEG", quality=65)
            paths.append(path)
            log.debug("Crop %s: %d bytes", path, os.path.getsize(path))

    return paths


# ── API submission ─────────────────────────────────────────────────────────
def submit_crops(crop_paths: list[str], session_id: str) -> str:
    """Returns job_id. Raises on non-retryable errors."""
    token = get_id_token()
    files = [("crops", (Path(p).name, open(p, "rb"), "image/jpeg")) for p in crop_paths]
    data  = {"device_id": DEVICE_ID, "session_id": session_id}

    resp = httpx.post(
        f"{API_URL}/v1/submit",
        headers={"Authorization": f"Bearer {token}"},
        files=files,
        data=data,
        timeout=30,
    )

    if resp.status_code == 401:
        # Token may have expired mid-flight — retry once with fresh token
        token = get_id_token(force=True)
        resp  = httpx.post(
            f"{API_URL}/v1/submit",
            headers={"Authorization": f"Bearer {token}"},
            files=files,
            data=data,
            timeout=30,
        )

    resp.raise_for_status()
    return resp.json()["job_id"]


# ── Result polling ─────────────────────────────────────────────────────────
def poll_result(job_id: str) -> dict:
    token    = get_id_token()
    deadline = time.time() + POLL_MAX_WAIT
    interval = float(POLL_INTERVAL)

    while time.time() < deadline:
        try:
            resp = httpx.get(
                f"{API_URL}/v1/result/{job_id}",
                headers={"Authorization": f"Bearer {token}"},
                timeout=10,
            )
            if resp.status_code == 404:
                time.sleep(interval)
                interval = min(interval * 1.4, 15)
                continue
            resp.raise_for_status()
            data = resp.json()
            if data["status"] in ("completed", "failed"):
                return data
        except Exception as exc:
            log.warning("Poll error: %s", exc)
        time.sleep(interval)
        interval = min(interval * 1.4, 15)

    raise TimeoutError(f"Job {job_id} did not complete in {POLL_MAX_WAIT}s")


# ── Result printer ─────────────────────────────────────────────────────────
def print_result(result: dict):
    print("\n" + "=" * 60)
    print("MEDIBOX RESULT")
    print("=" * 60)
    if result.get("status") == "failed":
        print(f"Status : FAILED")
        print(f"Error  : {result.get('error', 'unknown')}")
    else:
        meds = result.get("result", {}).get("medications", [])
        conf = result.get("result", {}).get("overall_confidence", 0)
        print(f"Status     : COMPLETED")
        print(f"Confidence : {conf:.0%}")
        print(f"Medications: {len(meds)}")
        print("-" * 60)
        for i, med in enumerate(meds, 1):
            print(f"  {i}. {med.get('name', '?')}")
            print(f"     Dose    : {med.get('dosage', '?')}")
            print(f"     Duration: {med.get('duration', '?')}")
            print(f"     Conf    : {med.get('confidence', 0):.0%}")
    print("-" * 60)
    print(f"⚠  {result.get('disclaimer', 'Pharmacist verification required.')}")
    print("=" * 60 + "\n")


# ── Main ───────────────────────────────────────────────────────────────────
def main():
    logging.basicConfig(
        level=logging.INFO,
        format='{"ts":"%(asctime)s","level":"%(levelname)s","msg":"%(message)s",'
               '"device":"' + DEVICE_ID + '"}',
        datefmt="%Y-%m-%dT%H:%M:%SZ",
    )

    parser = argparse.ArgumentParser()
    parser.add_argument("--image", help="Path to image file (skip camera capture)")
    parser.add_argument("--queue", action="store_true",
                        help="Submit via offline queue instead of direct submit")
    args = parser.parse_args()

    init_db()

    with tempfile.TemporaryDirectory() as tmp:
        # 1. Capture or load image
        img_path = args.image or os.path.join(tmp, "prescription.jpg")
        if not args.image:
            print("Capturing image...")
            capture_image(img_path)

        # 2. Slice into grid
        print("Slicing into crops...")
        crop_dir  = os.path.join(tmp, "crops")
        os.makedirs(crop_dir)
        crops     = slice_into_crops(img_path, crop_dir)
        print(f"  {len(crops)} crops ready")

        session_id = str(uuid.uuid4())

        if args.queue:
            # Save to offline queue (worker loop will submit)
            enqueue(crop_dir, DEVICE_ID)
            print(f"Queued for offline submission. session_id={session_id}")
            return

        # 3. Submit directly
        print("Submitting to Medibox API...")
        try:
            job_id = submit_crops(crops, session_id)
            print(f"  Job ID: {job_id}")
        except httpx.ConnectError:
            print("Network unreachable — saving to offline queue")
            enqueue(crop_dir, DEVICE_ID)
            return
        except Exception as exc:
            print(f"Submit failed: {exc}")
            sys.exit(1)

        # 4. Poll for result
        print("Waiting for result...")
        result = poll_result(job_id)
        print_result(result)


if __name__ == "__main__":
    main()
PYEOF

================================================================================
TROUBLESHOOTING
================================================================================

"Firebase token verification failed"
  → Check MEDIBOX_FIREBASE_API_KEY in edge.env is the Web API Key (AIzaSy...)
  → Check the test user exists in Firebase Auth console
  → Check NTP: timedatectl status must show synchronized: yes

"Connection refused" / "Network error"
  → MEDIBOX_API_URL is wrong or deploy hasn't run yet
  → Check: curl -s "$MEDIBOX_API_URL/v1/healthz"

"Permission denied on /etc/medibox/medibox-edge.json"
  → sudo chown medibox:medibox /etc/medibox/medibox-edge.json
  → sudo chmod 600 /etc/medibox/medibox-edge.json

"Module not found: auth" or "No module named picamera2"
  → Use /opt/medibox/venv/bin/python not system python3
  → If picamera2 missing: sudo /opt/medibox/venv/bin/pip install picamera2

View live logs:
  journalctl -u medibox-edge -f

Manually run as medibox user:
  sudo -u medibox -E /opt/medibox/venv/bin/python /opt/medibox/submit_prescription.py

Check offline queue:
  sqlite3 /var/lib/medibox/queue.db \
    "SELECT session_id, status, datetime(created_at,'unixepoch') FROM jobs ORDER BY id DESC LIMIT 10;"
